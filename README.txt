def normalizar(
    texto
):

    texto = str(
        texto
        or
        ""
    ).lower()


    texto = unicodedata.normalize(
        "NFD",
        texto
    )


    texto = "".join(
        caracter
        for caracter in texto
        if unicodedata.category(
            caracter
        )
        !=
        "Mn"
    )


    return texto.strip()


def limitar(
    valor,
    minimo,
    maximo
):

    return max(
        minimo,
        min(
            valor,
            maximo
        )
    )


def separar_oraciones(
    texto
):

    return [
        elemento.strip()

        for elemento
        in re.split(
            r"(?<=[\.\!\?])\s+|\n+",
            texto
        )

        if elemento.strip()
    ]


def normalizar_username(
    username
):

    return normalizar(
        username
    ).replace(
        " ",
        ""
    )

def inicializar_db():

    with conectar_db() as conexion:

        conexion.execute(
            """
            CREATE SCHEMA IF NOT EXISTS portal
            """
        )


        conexion.execute(
            """
            CREATE TABLE IF NOT EXISTS portal.usuarios (

                id BIGINT
                    GENERATED BY DEFAULT AS IDENTITY
                    PRIMARY KEY,

                username TEXT
                    UNIQUE
                    NOT NULL,

                nombre_visible TEXT
                    NOT NULL,

                password_salt BYTEA
                    NOT NULL,

                password_hash BYTEA
                    NOT NULL,

                creado_en TIMESTAMPTZ
                    NOT NULL
                    DEFAULT NOW()
            )
            """
        )


        conexion.execute(
            """
            CREATE TABLE IF NOT EXISTS portal.perfiles (

                user_id BIGINT
                    PRIMARY KEY
                    REFERENCES portal.usuarios(id)
                    ON DELETE CASCADE,

                cv_texto TEXT
                    NOT NULL
                    DEFAULT '',

                actualizado_en TIMESTAMPTZ
                    NOT NULL
                    DEFAULT NOW()
            )
            """
        )


        conexion.execute(
            """
            CREATE TABLE IF NOT EXISTS portal.preferencias (

                user_id BIGINT
                    PRIMARY KEY
                    REFERENCES portal.usuarios(id)
                    ON DELETE CASCADE,

                cargo_deseado TEXT
                    NOT NULL
                    DEFAULT 'analista',

                modalidad TEXT
                    NOT NULL
                    DEFAULT 'hibrida',

                sueldo_minimo BIGINT
                    NOT NULL
                    DEFAULT 1200000,

                acepta_presencial BOOLEAN
                    NOT NULL
                    DEFAULT FALSE,

                ciudad TEXT
                    NOT NULL
                    DEFAULT 'Santiago',

                actualizado_en TIMESTAMPTZ
                    NOT NULL
                    DEFAULT NOW()
            )
            """
        )


        conexion.execute(
            """
            CREATE TABLE IF NOT EXISTS portal.interacciones (

                id BIGINT
                    GENERATED BY DEFAULT AS IDENTITY
                    PRIMARY KEY,

                user_id BIGINT
                    NOT NULL
                    REFERENCES portal.usuarios(id)
                    ON DELETE CASCADE,

                accion TEXT
                    NOT NULL,

                oferta_nombre TEXT,

                cargo TEXT,

                modalidad TEXT,

                ciudad TEXT,

                sueldo BIGINT,

                habilidades_json JSONB
                    NOT NULL
                    DEFAULT '[]'::jsonb,

                motivo TEXT,

                source_key TEXT,

                creado_en TIMESTAMPTZ
                    NOT NULL
                    DEFAULT NOW()
            )
            """
        )


        conexion.execute(
            """
            ALTER TABLE portal.interacciones
            ADD COLUMN IF NOT EXISTS source_key TEXT
            """
        )


        # --------------------------------------------------------------
        # AUDITAR ÍNDICE
        # --------------------------------------------------------------

        indice = conexion.execute(
            """
            SELECT indexdef

            FROM pg_indexes

            WHERE schemaname = 'portal'
              AND tablename = 'interacciones'
              AND indexname = 'idx_portal_interacciones_source'
            """

        reconstruir_indice = False


        if indice is None:

        else:

            definicion = indice[

            if (
                "CREATE UNIQUE INDEX"

                in definicion
            ):

        if reconstruir_indice:

                """

            )


            conexion.execute(
                """
                CREATE UNIQUE INDEX
                    idx_portal_interacciones_source

                ON portal.interacciones(source_key)

            )


        conexion.execute(
            """
            CREATE INDEX IF NOT EXISTS
                idx_portal_interacciones_user

            """
        )

def password_valida(
    password
):

    if len(
        password
    ) < 8:

        return (
            False,
            "La contraseña debe tener al menos 8 caracteres."
        )


    if not re.search(
        r"[A-Za-zÁÉÍÓÚáéíóúÑñ]",
        password
    ):

        return (
            False,
            "Debe contener al menos una letra."
        )


    if not re.search(
        r"\d",
        password
    ):

        return (
            False,
            "Debe contener al menos un número."
        )


    return (
        True,
        None
    )


def hash_password(
    password,
    salt=None
):

    if salt is None:

        salt = os.urandom(
            16
        )


    password_hash = hashlib.scrypt(
        password.encode(
            "utf-8"
        ),
        salt=salt,
        n=2**14,
        r=8,
        p=1,
        dklen=64
    )


    return (
        salt,
        password_hash
    )


def verificar_password(
    password,
    salt,
    password_hash
):

    _, candidato = hash_password(
        password,
        bytes(
            salt
        )
    )


    return hmac.compare_digest(
        candidato,
        bytes(
            password_hash
        )
    )

def crear_usuario(
    username,
    nombre_visible,
    password
):

    username = normalizar_username(
        username
    )


    nombre_visible = str(
        nombre_visible
    ).strip()


    if not re.fullmatch(
        r"[a-z0-9._-]{3,32}",
        username
    ):

        return (
            False,
            "Usuario inválido.",
            None
        )


    if not nombre_visible:

        return (
            False,
            "Ingresa tu nombre.",
            None
        )


    valida, mensaje = password_valida(
        password
    )


    if not valida:

        return (
            False,
            mensaje,
            None
        )


    salt, password_hash = hash_password(
        password
    )


    with conectar_db() as conexion:

        fila = conexion.execute(
            """
            INSERT INTO portal.usuarios (
                username,
                nombre_visible,
                password_salt,
                password_hash
            )

            VALUES (
                %s,
                %s,
                %s,
                %s
            )

            ON CONFLICT(username)
            DO NOTHING

            RETURNING id
            """,
            (
                username,
                nombre_visible,
                salt,
                password_hash
            )
        ).fetchone()


        if fila is None:

            return (
                False,
                "Ese usuario ya existe.",
                None
            )


        user_id = fila[
            "id"
        ]


        conexion.execute(
            """
            INSERT INTO portal.perfiles (
                user_id,
                cv_texto
            )

            VALUES (
                %s,
                ''
            )

            ON CONFLICT(user_id)
            DO NOTHING
            """,
            (
                user_id,
            )
        )


        conexion.execute(
            """
            INSERT INTO portal.preferencias (
                user_id,
                cargo_deseado,
                modalidad,
                sueldo_minimo,
                acepta_presencial,
                ciudad
            )

            VALUES (
                %s,
                'analista',
                'hibrida',
                1200000,
                FALSE,
                'Santiago'
            )

            ON CONFLICT(user_id)
            DO NOTHING
            """,
            (
                user_id,
            )
        )


    return (
        True,
        "Cuenta creada.",
        user_id
    )


def autenticar_usuario(
    username,
    password
):

    username = normalizar_username(
        username
    )


    with conectar_db() as conexion:

        fila = conexion.execute(
            """
            SELECT
                id,
                username,
                nombre_visible,
                password_salt,
                password_hash

            FROM portal.usuarios

            WHERE username = %s
            """,
            (
                username,
            )
        ).fetchone()


    if fila is None:

        return None


    if not verificar_password(
        password,
        fila[
            "password_salt"
        ],
        fila[
            "password_hash"
        ]
    ):

        return None


    return {

        "id":
            fila[
                "id"
            ],

        "username":
            fila[
                "username"
            ],

        "nombre_visible":
            fila[
                "nombre_visible"
            ]
    }

        dict(

def actualizar_password(
    user_id,
    nueva_password
):

    valida, mensaje = password_valida(
        nueva_password
    )


    if not valida:

        return (
            False,
            mensaje
        )


    salt, password_hash = hash_password(
        nueva_password
    )


    with conectar_db() as conexion:

        resultado = conexion.execute(
            """
            UPDATE portal.usuarios

            SET
                password_salt = %s,
                password_hash = %s

            WHERE id = %s
            """,
            (
                salt,
                password_hash,
                user_id
            )
        )


    return (
        resultado.rowcount
        ==
        1,
        (
            "Contraseña actualizada."
            if resultado.rowcount
            ==
            1
            else
            "No se pudo actualizar."
        )
    )

def guardar_interaccion(
    user_id,
    evento,
    source_key=None
):

    with conectar_db() as conexion:

        if source_key is None:

            conexion.execute(
                """
                INSERT INTO portal.interacciones (
                    user_id,
                    accion,
                    oferta_nombre,
                    cargo,
                    modalidad,
                    ciudad,
                    sueldo,
                    habilidades_json,
                    motivo
                )

                VALUES (
                    %s,
                    %s,
                    %s,
                    %s,
                    %s,
                    %s,
                    %s,
                    %s::jsonb,
                    %s
                )
                """,
                (
                    user_id,
                    evento.get(
                        "accion"
                    ),
                    evento.get(
                        "oferta_nombre"
                    ),
                    evento.get(
                        "cargo"
                    ),
                    evento.get(
                        "modalidad"
                    ),
                    evento.get(
                        "ciudad"
                    ),
                    evento.get(
                        "sueldo"
                    ),
                    json.dumps(
                        evento.get(
                            "habilidades",
                            []
                        ),
                        ensure_ascii=False
                    ),
                    evento.get(
                        "motivo"
                    )
                )
            )

            return True


        resultado = conexion.execute(
            """
            INSERT INTO portal.interacciones (
                user_id,
                accion,
                oferta_nombre,
                cargo,
                modalidad,
                ciudad,
                sueldo,
                habilidades_json,
                motivo,
                source_key
            )

            VALUES (
                %s,
                %s,
                %s,
                %s,
                %s,
                %s,
                %s,
                %s::jsonb,
                %s,
                %s
            )

            ON CONFLICT(source_key)
            DO NOTHING
            """,
            (
                user_id,
                evento.get(
                    "accion"
                ),
                evento.get(
                    "oferta_nombre"
                ),
                evento.get(
                    "cargo"
                ),
                evento.get(
                    "modalidad"
                ),
                evento.get(
                    "ciudad"
                ),
                evento.get(
                    "sueldo"
                ),
                json.dumps(
                    evento.get(
                        "habilidades",
                        []
                    ),
                    ensure_ascii=False
                ),
                evento.get(
                    "motivo"
                ),
                source_key
            )
        )


        return (
            resultado.rowcount
            ==
            1
        )


def cargar_interacciones(
    user_id
):

    with conectar_db() as conexion:

        filas = conexion.execute(
            """
            SELECT
                id,
                accion,
                oferta_nombre,
                cargo,
                modalidad,
                ciudad,
                sueldo,
                habilidades_json,
                motivo,
                creado_en

            FROM portal.interacciones

            WHERE user_id = %s

            ORDER BY id ASC
            """,
            (
                user_id,
            )
        ).fetchall()


    resultado = []


    for fila in filas:

        habilidades = fila[
            "habilidades_json"
        ]


        if isinstance(
            habilidades,
            str
        ):

            habilidades = json.loads(
                habilidades
            )


        resultado.append(
            {

                "id":
                    fila[
                        "id"
                    ],

                "accion":
                    fila[
                        "accion"
                    ],

                "oferta_nombre":
                    fila[
                        "oferta_nombre"
                    ],

                "cargo":
                    fila[
                        "cargo"
                    ],

                "modalidad":
                    fila[
                        "modalidad"
                    ],

                "ciudad":
                    fila[
                        "ciudad"
                    ],

                "sueldo":
                    fila[
                        "sueldo"
                    ],

                "habilidades":
                    list(
                        habilidades
                        or
                        []
                    ),

                "motivo":
                    fila[
                        "motivo"
                    ],

                "creado_en":
                    fila[
                        "creado_en"
                    ]
            }
        )


    return resultado


def borrar_interacciones(
    user_id
):

    with conectar_db() as conexion:

        conexion.execute(
            """
            DELETE FROM portal.interacciones

            WHERE user_id = %s
            """,
            (
                user_id,
            )
        )

def alias_presente(
    texto,
    alias
):

    return re.search(
        r"(?<!\w)"
        +
        re.escape(
            normalizar(
                alias
            )
        )
        +
        r"(?!\w)",
        normalizar(
            texto
        )
    ) is not None


def contexto_negado(
    oracion,
    alias
):

    texto = normalizar(
        oracion
    )


    habilidad = normalizar(
        alias
    )


    patrones = [

        rf"\bno\s+(?:manejo|domino|conozco|utilizo|uso)\s+"
        rf"(?:actualmente\s+)?{re.escape(habilidad)}\b",

        rf"\bno\s+tengo\s+(?:experiencia|conocimientos?)\s+"
        rf"(?:en|de|con)?\s*{re.escape(habilidad)}\b",

        rf"\bsin\s+(?:experiencia|conocimientos?|manejo)\s+"
        rf"(?:en|de|con)?\s*{re.escape(habilidad)}\b",

        rf"\b(?:quiero|quisiera|deseo|me\s+interesa)\s+"
        rf"aprender\s+{re.escape(habilidad)}\b",

        rf"\baprender\s+{re.escape(habilidad)}\b"
    ]


    return any(
        re.search(
            patron,
            texto
        )

        for patron
        in patrones
    )


def detectar_nivel_local(
    oracion,
    alias,
    cantidad
):

    texto = normalizar(
        oracion
    )


    habilidad = normalizar(
        alias
    )


    patrones = [

        rf"{re.escape(habilidad)}\s*"
        rf"(?:nivel\s*)?{NIVEL_RE}",

        rf"{NIVEL_RE}\s+"
        rf"(?:en\s+)?{re.escape(habilidad)}",

        rf"(?:nivel|manejo|conocimientos?|experiencia)\s+"
        rf"{NIVEL_RE}\s+"
        rf"(?:en|de|con)?\s*"
        rf"{re.escape(habilidad)}"
    ]


    for patron in patrones:

        coincidencia = re.search(
            patron,
            texto
        )


        if coincidencia:

            for grupo in coincidencia.groups():

                if grupo in NIVELES:

                    return (
                        NIVELES[
                            grupo
                        ],
                        100
                    )


    if cantidad == 1:

        coincidencia = re.search(
            NIVEL_RE,
            texto
        )


        if coincidencia:

            return (
                NIVELES[
                    coincidencia.group(
                        1
                    )
                ],
                90
            )


    return (
        3,
        70
    )


def extraer_habilidades_cv(
    texto
):

    encontradas = {}

    negadas = {}


    for oracion in separar_oraciones(
        texto
    ):

        presentes = []


        for habilidad, aliases in CATALOGO_HABILIDADES.items():

            alias_encontrado = None


            for alias in sorted(
                aliases,
                key=len,
                reverse=True
            ):

                if not alias_presente(
                    oracion,
                    alias
                ):

                    continue


                if (
                    normalizar(
                        alias
                    )
                    ==
                    "bi"
                    and
                    re.search(
                        r"power\s+bi",
                        normalizar(
                            oracion
                        )
                    )
                ):

                    continue


                alias_encontrado = alias

                break


            if alias_encontrado:

                presentes.append(
                    (
                        habilidad,
                        alias_encontrado
                    )
                )


        for habilidad, alias in presentes:

            if contexto_negado(
                oracion,
                alias
            ):

                negadas.setdefault(
                    habilidad,
                    []
                ).append(
                    oracion
                )

                continue


            nivel, confianza = detectar_nivel_local(
                oracion,
                alias,
                len(
                    presentes
                )
            )


            if habilidad not in encontradas:

                encontradas[
                    habilidad
                ] = {

                    "nivel":
                        nivel,

                    "confianza":
                        confianza,

                    "evidencias": [
                        oracion
                    ]
                }

            else:

                encontradas[
                    habilidad
                ][
                    "evidencias"
                ].append(
                    oracion
                )


                # --------------------------------------------------
                # V0.16.3.1
                # La evidencia posterior más fuerte también debe
                # actualizar la confianza, no sólo el nivel.
                # --------------------------------------------------

                encontradas[
                    habilidad
                ][
                    "confianza"
                ] = max(
                    encontradas[
                        habilidad
                    ][
                        "confianza"
                    ],
                    confianza
                )


                if (
                    nivel
                    >
                    encontradas[
                        habilidad
                    ][
                        "nivel"
                    ]
                ):

                    encontradas[
                        habilidad
                    ][
                        "nivel"
                    ] = nivel


    return (
        encontradas,
        negadas
    )


def extraer_campo_linea(
    texto,
    etiquetas
):

    for linea in texto.splitlines():

        linea = linea.strip()


        for etiqueta in etiquetas:

            coincidencia = re.match(
                rf"(?i)^\s*"
                rf"{re.escape(etiqueta)}"
                rf"\s*:\s*(.+?)\s*$",
                linea
            )


            if coincidencia:

                return coincidencia.group(
                    1
                ).strip()


    return None


def procesar_cv(
    texto
):

    nombre = extraer_campo_linea(
        texto,
        [
            "nombre",
            "nombre completo"
        ]
    )


    profesion = extraer_campo_linea(
        texto,
        [
            "profesion",
            "profesión",
            "titulo profesional",
            "título profesional"
        ]
    )


    ciudad = extraer_campo_linea(
        texto,
        [
            "ciudad",
            "residencia",
            "ubicacion",
            "ubicación"
        ]
    )


    if (
        ciudad
        and
        "santiago"
        in normalizar(
            ciudad
        )
    ):

        ciudad = "Santiago"


    experiencia = None


    for patron in [

        r"\b(?:tengo|cuento\s+con|con)\s+"
        r"(\d+)\s*(?:anos|ano)\s+de\s+experiencia\b",

        r"\b(\d+)\s*(?:anos|ano)\s+"
        r"de\s+experiencia\s+"
        r"(?:profesional|laboral)\b"
    ]:

        coincidencia = re.search(
            patron,
            normalizar(
                texto
            )
        )


        if coincidencia:

            experiencia = int(
                coincidencia.group(
                    1
                )
            )

            break


    detalle, negadas = extraer_habilidades_cv(
        texto
    )


    habilidades = {

        habilidad:
            datos[
                "nivel"
            ]

        for habilidad, datos
        in detalle.items()
    }


    campos = [
        nombre,
        profesion,
        ciudad,
        experiencia
    ]


    confianza_campos = (
        sum(
            campo is not None
            for campo in campos
        )
        /
        4
        *
        100
    )


    confianza_skills = (
        sum(
            datos[
                "confianza"
            ]
            for datos
            in detalle.values()
        )
        /
        len(
            detalle
        )

        if detalle
        else 0
    )


    confianza = round(
        confianza_campos
        *
        0.60
        +
        confianza_skills
        *
        0.40,
        2
    )


    return {

        "nombre":
            nombre,

        "profesion":
            profesion,

        "experiencia":
            experiencia,

        "ciudad":
            ciudad,

        "habilidades":
            habilidades,

        "detalle_habilidades":
            detalle,

        "negadas":
            negadas,

        "confianza":
            confianza
    }

def cumplimiento_nivel(
    nivel_candidato,
    nivel_requerido
):

    if nivel_requerido <= 0:

        return 100


    if nivel_candidato <= 0:

        return 0


    return min(
        100,
        nivel_candidato
        /
        nivel_requerido
        *
        100
    )


def matching_habilidades(
    requeridas,
    candidatas
):

    requeridas_ordenadas = sorted(
        requeridas.items(),
        key=lambda item: (
            -item[1][
                "importancia"
            ],
            -item[1][
                "nivel_requerido"
            ],
            item[0]
        )
    )


    skills_candidato = list(
        candidatas.keys()
    )


    mejor_score = -1

    mejor_asignacion = None


    def buscar(
        indice,
        usadas,
        score,
        asignacion
    ):

        nonlocal mejor_score
        nonlocal mejor_asignacion


        if indice == len(
            requeridas_ordenadas
        ):

            if score > mejor_score:

                mejor_score = score

                mejor_asignacion = list(
                    asignacion
                )

            return


        requerida, datos = requeridas_ordenadas[
            indice
        ]


        asignacion.append(
            (
                requerida,
                None,
                0,
                "SIN MATCH",
                0,
                0,
                0,
                datos
            )
        )


        buscar(
            indice + 1,
            usadas,
            score,
            asignacion
        )


        asignacion.pop()


        for candidata in skills_candidato:

            if candidata in usadas:

                continue


            relacion = relacion_habilidad(
                candidata,
                requerida
            )


            if relacion[
                "similitud"
            ] <= 0:

                continue


            nivel = cumplimiento_nivel(
                candidatas[
                    candidata
                ],
                datos[
                    "nivel_requerido"
                ]
            )


            tipo = relacion[
                "tipo"
            ]


            cumplimiento = (
                nivel
                *
                FACTOR_CUMPLIMIENTO[
                    tipo
                ]
            )


            potencial = (
                relacion[
                    "similitud"
                ]
                +
                nivel
            ) / 2


            aporte = (
                datos[
                    "importancia"
                ]
                *
                (
                    relacion[
                        "similitud"
                    ]
                    *
                    10
                    +
                    cumplimiento
                )
            )


            asignacion.append(
                (
                    requerida,
                    candidata,
                    relacion[
                        "similitud"
                    ],
                    tipo,
                    nivel,
                    cumplimiento,
                    potencial,
                    datos
                )
            )


            buscar(
                indice + 1,
                usadas
                |
                {
                    candidata
                },
                score
                +
                aporte,
                asignacion
            )


            asignacion.pop()


    buscar(
        0,
        set(),
        0,
        []
    )


    resultado = {}


    for (
        requerida,
        candidata,
        similitud,
        tipo,
        nivel,
        cumplimiento,
        potencial,
        datos
    ) in (
        mejor_asignacion
        or []
    ):

        resultado[
            requerida
        ] = {

            "candidato":
                candidata,

            "similitud":
                similitud,

            "tipo":
                tipo,

            "nivel":
                nivel,

            "cumplimiento":
                cumplimiento,

            "potencial":
                potencial,

            "importancia":
                datos[
                    "importancia"
                ],

            "nivel_requerido":
                datos[
                    "nivel_requerido"
                ]
        }


    return resultado


def score_habilidades(
    matches,
    campo
):

    if not matches:

        return 0


    peso = sum(
        datos[
            "importancia"
        ]
        for datos
        in matches.values()
    )


    if peso == 0:

        return 0


    return (
        sum(
            datos[
                campo
            ]
            *
            datos[
                "importancia"
            ]

            for datos
            in matches.values()
        )
        /
        peso
    )

def detectar_brechas(
    matches
):

    resultado = []


    for habilidad, datos in matches.items():

        if datos[
            "importancia"
        ] < 4:

            continue


        if datos[
            "candidato"
        ] is None:

            resultado.append(
                {

                    "habilidad":
                        habilidad,

                    "tipo":
                        "AUSENCIA",

                    "deficit":
                        100,

                    "relacion":
                        "SIN MATCH"
                }
            )

            continue


        if (
            habilidad
            in HERRAMIENTAS_ESPECIFICAS
            and
            datos[
                "tipo"
            ]
            not in [
                "EXACTO",
                "EQUIVALENTE"
            ]
        ):

            resultado.append(
                {

                    "habilidad":
                        habilidad,

                    "tipo":
                        "HERRAMIENTA_ESPECIFICA",

                    "deficit":
                        100
                        -
                        datos[
                            "cumplimiento"
                        ],

                    "relacion":
                        datos[
                            "tipo"
                        ]
                }
            )

            continue


        if datos[
            "tipo"
        ] in [
            "TRANSFERIBLE",
            "RELACIONADA"
        ]:

            resultado.append(
                {

                    "habilidad":
                        habilidad,

                    "tipo":
                        "SUSTITUCION",

                    "deficit":
                        100
                        -
                        datos[
                            "cumplimiento"
                        ],

                    "relacion":
                        datos[
                            "tipo"
                        ]
                }
            )

            continue


        if datos[
            "nivel"
        ] < 100:

            resultado.append(
                {

                    "habilidad":
                        habilidad,

                    "tipo":
                        "NIVEL",

                    "deficit":
                        100
                        -
                        datos[
                            "nivel"
                        ],

                    "relacion":
                        datos[
                            "tipo"
                        ]
                }
            )


    return resultado


def penalizacion_brechas(
    brechas
):

    total = 0


    for brecha in brechas:

        if brecha[
            "tipo"
        ] == "AUSENCIA":

            total += 20


        elif brecha[
            "tipo"
        ] == "HERRAMIENTA_ESPECIFICA":

            total += (
                15

                if brecha[
                    "relacion"
                ]
                ==
                "RELACIONADA"

                else 10
            )


        elif brecha[
            "tipo"
        ] == "SUSTITUCION":

            total += min(
                12,
                brecha[
                    "deficit"
                ]
                *
                0.12
            )


        elif brecha[
            "tipo"
        ] == "NIVEL":

            total += (
                brecha[
                    "deficit"
                ]
                *
                0.10
            )


    return min(
        total,
        40
    )

PESO_ACCION = {

    "postular":
        3,

    "guardar":
        2,

    "ver":
        0.25,

    "rechazar":
        -2
}


def aprender_comportamiento(
    historial
):

    cargos = defaultdict(
        float
    )

    modalidades = defaultdict(
        float
    )

    ciudades = defaultdict(
        float
    )

    habilidades = defaultdict(
        float
    )


    interacciones = 0


    for evento in historial:

        accion = normalizar(
            evento.get(
                "accion"
            )
        )


        if accion not in PESO_ACCION:

            continue


        interacciones += 1


        peso = PESO_ACCION[
            accion
        ]


        cargo = normalizar_cargo(
            evento.get(
                "cargo"
            )
        )


        modalidad = normalizar(
            evento.get(
                "modalidad"
            )
        )


        ciudad = normalizar(
            evento.get(
                "ciudad"
            )
        )


        motivo = normalizar(
            evento.get(
                "motivo"
            )
        )


        if peso > 0:

            cargos[
                cargo
            ] += peso


            modalidades[
                modalidad
            ] += peso


            ciudades[
                ciudad
            ] += peso


            for skill in evento.get(
                "habilidades",
                []
            ):

                habilidades[
                    canon_skill(
                        skill
                    )
                ] += peso


        elif accion == "rechazar":

            if motivo == "cargo":

                cargos[
                    cargo
                ] += peso


            elif motivo == "modalidad":

                modalidades[
                    modalidad
                ] += peso


            elif motivo == "ubicacion":

                ciudades[
                    ciudad
                ] += peso


    confianza = min(
        100,
        interacciones
        /
        8
        *
        100
    )


    return {

        "interacciones":
            interacciones,

        "confianza":
            round(
                confianza,
                2
            ),

        "habilitado":
            (
                interacciones >= 3
                and
                confianza >= 40
            ),

        "afinidad_cargos":
            dict(
                cargos
            ),

        "afinidad_modalidades":
            dict(
                modalidades
            ),

        "afinidad_ciudades":
            dict(
                ciudades
            ),

        "afinidad_habilidades":
            dict(
                habilidades
            )
    }

def _json_dict_v01880(
    valor
):

    if valor is None:

        return {}


    if isinstance(
        valor,
        dict
    ):

        return deepcopy(
            valor
        )


    if isinstance(
        valor,
        str
    ):

        try:

            convertido = json.loads(
                valor
            )


            if isinstance(
                convertido,
                dict
            ):

                return convertido


        except Exception:

            pass


    return {}


def _texto_limpio_v01880(
    valor
):

    return str(
        valor
        or
        ""
    ).strip()


def _numero_no_negativo_v01880(
    valor,
    default=0.0
):

    try:

        numero = float(
            valor
        )


        return max(
            0.0,
            numero
        )


    except Exception:

        return float(
            default
        )

def oferta_valida(
    oferta,
    candidato
):

    if (
        oferta[
            "modalidad"
        ]
        ==
        "presencial"
        and
        not candidato[
            "acepta_presencial"
        ]
    ):

        return (
            False,
            "Modalidad presencial no aceptada"
        )


    if (
        oferta[
            "ciudad"
        ]
        not in candidato[
            "ciudades_aceptadas"
        ]
    ):

        return (
            False,
            "Ciudad no aceptada"
        )


    if (
        oferta[
            "sueldo"
        ] is not None
        and
        oferta[
            "sueldo"
        ]
        <
        candidato[
            "sueldo_minimo"
        ]
    ):

        return (
            False,
            "Sueldo bajo el mínimo"
        )


    return (
        True,
        None
    )


def señal(
    valor
):

    return limitar(
        valor
        /
        5,
        -1,
        1
    )


def ajuste_comportamental(
    oferta,
    perfil
):

    aprendizaje = perfil[
        "aprendizaje"
    ]


    if not aprendizaje[
        "habilitado"
    ]:

        return 0


    cargo = señal(
        aprendizaje[
            "afinidad_cargos"
        ].get(
            normalizar_cargo(
                oferta[
                    "cargo"
                ]
            ),
            0
        )
    )


    modalidad = señal(
        aprendizaje[
            "afinidad_modalidades"
        ].get(
            normalizar(
                oferta[
                    "modalidad"
                ]
            ),
            0
        )
    )


    ciudad = señal(
        aprendizaje[
            "afinidad_ciudades"
        ].get(
            normalizar(
                oferta[
                    "ciudad"
                ]
            ),
            0
        )
    )


    señales_skills = []


    for skill in oferta[
        "habilidades"
    ]:

        valor = aprendizaje[
            "afinidad_habilidades"
        ].get(
            skill,
            0
        )


        if valor != 0:

            señales_skills.append(
                señal(
                    valor
                )
            )


    skills = (
        sum(
            señales_skills
        )
        /
        len(
            señales_skills
        )

        if señales_skills
        else 0
    )


    bruto = (
        cargo
        *
        0.80
        +
        modalidad
        *
        0.30
        +
        ciudad
        *
        0.20
        +
        skills
        *
        0.70
    )


    return limitar(
        bruto
        *
        aprendizaje[
            "confianza"
        ]
        /
        100,
        -2,
        2
    )


def aplicar_guardrails(
    score,
    cargo,
    brechas
):

    if cargo < 20:

        score = min(
            score,
            35
        )


    elif cargo < 50:

        score = min(
            score,
            55
        )


    if any(
        brecha[
            "tipo"
        ]
        ==
        "HERRAMIENTA_ESPECIFICA"

        for brecha
        in brechas
    ):

        score = min(
            score,
            75
        )


    return score

def _normalizar_exp_v01881(
    valor
):

    texto = str(
        valor
        or
        ""
    ).lower()


    texto = normalizar(
        texto
    )


    texto = re.sub(
        r"\s+",
        " ",
        texto
    )


    return texto.strip()


def _texto_experiencia_v01881(
    experiencia
):

    if not isinstance(
        experiencia,
        dict
    ):

        return ""


    partes = [

        experiencia.get(
            "cargo",
            ""
        ),

        experiencia.get(
            "area",
            ""
        ),

        " ".join(
            experiencia.get(
                "funciones",
                []
            )
            if isinstance(
                experiencia.get(
                    "funciones"
                ),
                list
            )
            else [
                str(
                    experiencia.get(
                        "funciones",
                        ""
                    )
                )
            ]
        ),

        " ".join(
            experiencia.get(
                "habilidades",
                []
            )
            if isinstance(
                experiencia.get(
                    "habilidades"
                ),
                list
            )
            else [
                str(
                    experiencia.get(
                        "habilidades",
                        ""
                    )
                )
            ]
        )
    ]


    return _normalizar_exp_v01881(
        " ".join(
            str(
                parte
            )
            for parte
            in partes
        )
    )


def _contiene_alguno_v01881(
    texto,
    palabras
):

    texto = _normalizar_exp_v01881(
        texto
    )


    return any(
        _normalizar_exp_v01881(
            palabra
        )
        in texto
        for palabra
        in palabras
    )


def _cantidad_coincidencias_v01881(
    texto,
    palabras
):

    texto = _normalizar_exp_v01881(
        texto
    )


    encontrados = set()


    for palabra in palabras:

        canon = _normalizar_exp_v01881(
            palabra
        )


        if canon and canon in texto:

            encontrados.add(
                canon
            )


    return len(
        encontrados
    )


def detectar_dominio_cargo_v01881(
    cargo
):

    cargo_norm = _normalizar_exp_v01881(
        cargo
    )


    mejor_dominio = None

    mejor_score = 0


    for dominio, reglas in DOMINIOS_CARGO_V01881.items():

        score = _cantidad_coincidencias_v01881(
            cargo_norm,
            reglas[
                "cargo"
            ]
        )


        if score > mejor_score:

            mejor_dominio = dominio

            mejor_score = score


    return mejor_dominio

        "cargo",

def evaluar_experiencia_contextual_v01880(
    candidato,
    oferta,
    perfil=None
):

    experiencias = []


    if isinstance(
        candidato,
        dict
    ):

        experiencias = candidato.get(
            "experiencias",
            []
        )


    if (
        not experiencias
        and
        isinstance(
            perfil,
            dict
        )
    ):

        experiencias = perfil.get(
            "experiencias",
            []
        )


    experiencias = normalizar_experiencias_v01880(
        experiencias
    )


    detalle = []


    acumulados = {

        "DIRECTA":
            0.0,

        "FUNCIONAL_RELEVANTE":
            0.0,

        "RELACIONADA":
            0.0,

        "TRANSFERIBLE_ALTO_VALOR":
            0.0,

        "TRANSFERIBLE":
            0.0,

        "SIN_RELACION_CLARA":
            0.0
    }


    total = 0.0

    equivalente = 0.0


    for experiencia in experiencias:

        años = float(
            experiencia.get(
                "años",
                0
            )
            or
            0
        )


        total += años


        clasificacion = clasificar_experiencia_v01881(
            experiencia,
            oferta
        )


        tipo = clasificacion[
            "clasificacion"
        ]


        peso = float(
            clasificacion.get(
                "peso",
                0
            )
            or
            0
        )


        if tipo in acumulados:

            acumulados[
                tipo
            ] += años


        equivalente += (
            años
            *
            peso
        )


        detalle.append(
            {

                "cargo":
                    experiencia.get(
                        "cargo",
                        ""
                    ),

                "empresa":
                    experiencia.get(
                        "empresa",
                        ""
                    ),

                "area":
                    experiencia.get(
                        "area",
                        ""
                    ),

                "años":
                    años,

                "clasificacion":
                    tipo,

                "peso":
                    peso,

                "años_equivalentes":
                    años
                    *
                    peso,

                "razon":
                    clasificacion.get(
                        "razon",
                        ""
                    ),

                "cargo_similitud":
                    clasificacion.get(
                        "cargo_similitud",
                        0
                    ),

                "dominio_objetivo":
                    clasificacion.get(
                        "dominio_objetivo"
                    )
            }
        )


    return {

def score_experiencia_contextual_v01882(
    detalle,
    oferta
):

    if not isinstance(
        detalle,
        dict
    ):

        return 0.0


    if (
        detalle.get(
            "cantidad_experiencias",
            0
        )
        <=
        0
    ):

        return 0.0


    equivalente = float(
        detalle.get(
            "equivalente",
            0
        )
        or
        0
    )


    requerida = 0.0


    if isinstance(
        oferta,
        dict
    ):

        try:

            requerida = float(
                oferta.get(
                    "experiencia",
                    0
                )
                or
                0
            )


        except Exception:

            requerida = 0.0


    if requerida <= 0:

        # --------------------------------------------------------------
        # Si la oferta no exige años concretos, la existencia de
        # experiencia contextual relevante sigue siendo positiva.
        # --------------------------------------------------------------

        if equivalente > 0:

            return 100.0


        return 50.0


    return limitar(
        equivalente
        /
        requerida
        *
        100,
        0,
        100
    )


def limite_contextual_por_cargo_v01882(
    evaluacion_base
):

    # ------------------------------------------------------------------
    # V0.18.8.2 original:
    # la experiencia contextual nunca puede alterar el resultado
    # en más de 8 puntos.
    #
    # Conservamos el helper por separado porque posteriormente podremos
    # refinar el límite según el tipo de transición profesional.
    # ------------------------------------------------------------------

    return 8.0


def ajuste_experiencia_contextual_v01882(
    score_contextual,
    detalle,
    evaluacion_base
):

    if not isinstance(
        detalle,
        dict
    ):

        return 0.0


    if (
        detalle.get(
            "cantidad_experiencias",
            0
        )
        <=
        0
    ):

        # --------------------------------------------------------------
        # Sin experiencias estructuradas no penalizamos.
        # Compatibilidad histórica intacta.
        # --------------------------------------------------------------

        return 0.0


    limite_ajuste = limite_contextual_por_cargo_v01882(
        evaluacion_base
    )


    # ------------------------------------------------------------------
    # 50 = punto neutro.
    # 100 = +8.
    #   0 = -8.
    # ------------------------------------------------------------------

    ajuste = (
        (
            float(
                score_contextual
            )
            -
            50.0
        )
        /
        50.0
        *
        limite_ajuste
    )


    return limitar(
        ajuste,
        -limite_ajuste,
        limite_ajuste
    )


def razon_experiencia_contextual_v01882(
    detalle
):

    if not isinstance(
        detalle,
        dict
    ):

        return (
            "No existe información estructurada suficiente."
        )


    if detalle.get(
        "directa",
        0
    ) > 0:

        return (
            "El perfil contiene experiencia directamente relacionada "
            "con el cargo."
        )


    if detalle.get(
        "funcional_relevante",
        0
    ) > 0:

        return (

            "aunque el nombre del cargo anterior sea diferente."
        )


    if detalle.get(
        "relacionada",
        0
    ) > 0:

        return (
            "El perfil contiene experiencia relacionada con el cargo."
        )


    if detalle.get(
        "transferible_alto_valor",
        0
    ) > 0:

        return (
            "El perfil contiene experiencia transferible de alto valor."
        )


    if detalle.get(
        "transferible",
        0
    ) > 0:

        return (
            "El perfil contiene competencias profesionales transferibles."
        )


    if detalle.get(
        "cantidad_experiencias",
        0
    ) > 0:

        return (
            "Existen experiencias registradas, pero no detectamos "
            "una relación profesional clara con este cargo."
        )


    return (
        "No existen experiencias estructuradas suficientes para "
        "aplicar ajuste contextual."
    )

def evaluar_oferta_base_v01882(
    oferta,
    candidato,
    perfil
):

    matches = matching_habilidades(
        oferta[
            "habilidades"
        ],
        candidato[
            "habilidades"
        ]
    )


    cumplimiento = score_habilidades(
        matches,
        "cumplimiento"
    )


    potencial = score_habilidades(
        matches,
        "potencial"
    )


    cargo = similitud_cargo(
        oferta[
            "cargo"
        ],
        candidato[
            "cargo_deseado"
        ]
    )


    experiencia = (
        min(
            100,
            candidato[
                "experiencia"
            ]
            /
            oferta[
                "experiencia"
            ]
            *
            100
        )

        if (
            candidato[
                "experiencia"
            ]
            is not None
            and
            oferta[
                "experiencia"
            ]
        )

        else 0
    )


    sueldo = (
        100

        if oferta[
            "sueldo"
        ]
        >=
        candidato[
            "sueldo_minimo"
        ]

        else
        oferta[
            "sueldo"
        ]
        /
        max(
            candidato[
                "sueldo_minimo"
            ],
            1
        )
        *
        100
    )


    brechas = detectar_brechas(
        matches
    )


    penalizacion = penalizacion_brechas(
        brechas
    )


    base = (
        cumplimiento
        *
        4
        +
        cargo
        *
        3
        +
        sueldo
        *
        2
        +
        experiencia
        *
        5
    ) / 14


    base = limitar(
        base
        -
        penalizacion,
        0,
        100
    )


    conducta = ajuste_comportamental(
        oferta,
        perfil
    )


    final = limitar(
        base
        +
        conducta,
        0,
        100
    )


    final = aplicar_guardrails(
        final,
        cargo,
        brechas
    )


    return {

        "final":
            final,

        "base":
            base,

        "conducta":
            conducta,

        "cumplimiento":
            cumplimiento,

        "potencial":
            potencial,

        "cargo":
            cargo,

        "cargo_clasificacion":
            clasificar_cargo(
                cargo
            ),

        "brechas":
            brechas,

        "matches":
            matches
    }

def _normalizar_clave_equidad_v01883(
    clave
):

    texto = str(
        clave
        or
        ""
    ).strip().lower()


    reemplazos = {

        "á":
            "a",

        "é":
            "e",

        "í":
            "i",

        "ó":
            "o",

        "ú":
            "u",

        "ü":
            "u",

        "ñ":
            "n"
    }


    for origen, destino in reemplazos.items():

        texto = texto.replace(
            origen,
            destino
        )


    texto = re.sub(
        r"[^a-z0-9]+",
        "_",
        texto
    )


    return texto.strip(
        "_"
    )

}


def sanitizar_para_score_v01883(
    valor
):

    # ------------------------------------------------------------------
    # Diccionarios:
    #
    # eliminamos claves sensibles recursivamente.
    # ------------------------------------------------------------------

    if isinstance(
        valor,
        dict
    ):

        limpio = {}


        for clave, contenido in valor.items():

            clave_norm = _normalizar_clave_equidad_v01883(
                clave
            )


            if clave_norm in ATRIBUTOS_NO_SCORE_NORMALIZADOS_V01883:

                continue


            limpio[
                clave
            ] = sanitizar_para_score_v01883(
                contenido
            )


        return limpio


    # ------------------------------------------------------------------
    # Listas
    # ------------------------------------------------------------------

    if isinstance(
        valor,
        list
    ):

        return [

            sanitizar_para_score_v01883(
                item
            )

            for item
            in valor
        ]


    # ------------------------------------------------------------------
    # Tuplas
    # ------------------------------------------------------------------

    if isinstance(
        valor,
        tuple
    ):

        return tuple(

            sanitizar_para_score_v01883(
                item
            )

            for item
            in valor
        )


    # ------------------------------------------------------------------
    # Sets
    # ------------------------------------------------------------------

    if isinstance(
        valor,
        set
    ):

        return {

            sanitizar_para_score_v01883(
                item
            )

            for item
            in valor
        }


    # ------------------------------------------------------------------
    # Escalares
    # ------------------------------------------------------------------

    return valor

        score,

def construir_ranking(
    candidato,
    perfil
):

    ranking = []

    descartadas = []


    for nombre, oferta in OFERTAS.items():

        valida, motivo = oferta_valida(
            oferta,
            candidato
        )


        if not valida:

            descartadas.append(
                (
                    nombre,
                    motivo
                )
            )

            continue


        ranking.append(
            {

                "nombre":
                    nombre,

                "oferta":
                    oferta,

                "evaluacion":
                    evaluar_oferta(
                        oferta,
                        candidato,
                        perfil
                    )
            }
        )


    ranking.sort(
        key=lambda item:
            item[
                "evaluacion"
            ][
                "final"
            ],
        reverse=True
    )


    return (
        ranking,
        descartadas
    )

            )


            registrar = st.form_submit_button(

def _valor_match_v018842(
    match,
    claves,
    default=None
):

    if not isinstance(
        match,
        dict
    ):

        return default


    for clave in claves:

        if clave in match:

            return match[
                clave
            ]


    return default

def _nivel_explicito_requerido_v018842(
    match
):

    # ------------------------------------------------------------------
    # La prioridad es metadata explícita.
    #
    # Esto evita mentir al usuario:
    #
    # si el parser asignó nivel 3 por defecto pero la oferta jamás dijo
    # "intermedio", mostramos:
    #
    #   Nivel solicitado no especificado
    #
    # y NO:
    #
    #   Nivel solicitado: Intermedio
    # ------------------------------------------------------------------

    directo = _valor_match_v018842(
        match,
        [
            "nivel_explicito",
            "nivel_requerido_explicito"
        ],
        None
    )


    if directo is not None:

        return bool(
            directo
        )


    requerida = _valor_match_v018842(
        match,
        [
            "requerida",
            "habilidad_requerida"
        ],
        {}
    )


    if isinstance(
        requerida,
        dict
    ):

        if "nivel_explicito" in requerida:

            return bool(
                requerida[
                    "nivel_explicito"
                ]
            )


        # --------------------------------------------------------------
        # Compatibilidad con parsers antiguos.
        #
        # Si existe texto original y contiene un descriptor de nivel,
        # podemos inferir que fue explícito.
        # --------------------------------------------------------------

        texto = str(
            requerida.get(
                "texto",
                ""
            )
            or
            requerida.get(
                "origen",
                ""
            )
            or
            ""
        ).lower()


        patrones = [

            "basico",
            "básico",

            "inicial",

            "intermedio",

            "avanzado",

            "experto",

            "senior",
            "sénior",

            "junior",
            "júnior"
        ]


        if texto:

            return any(
                patron
                in texto
                for patron
                in patrones
            )


    # --------------------------------------------------------------
    # Sin evidencia de que el nivel haya sido escrito por la empresa,
    # asumimos NO explícito a nivel de presentación.
    # --------------------------------------------------------------

    return False

def estado_conocimiento_v018842(

        match
    )


    nivel_explicito = _nivel_explicito_requerido_v018842(
        match
    )


    if nivel_candidato <= 0:

        return {

            "codigo":
                "NO_DETECTADO",

            "icono":
                "⚠",

            "texto":
                "Nivel no detectado",

            "tono":
                "warning",

                nivel_candidato,

            "nivel_requerido":
                nivel_requerido,

            "nivel_explicito":
                nivel_explicito
        }


    # ------------------------------------------------------------------
    # Si la oferta no especificó nivel, no podemos afirmar que el
    # candidato esté por debajo de un nivel que la empresa no publicó.
    # ------------------------------------------------------------------

    if not nivel_explicito:

        return {

            "codigo":
                "NIVEL_NO_ESPECIFICADO",

            "icono":
                "⚠",

            "texto":
                "Nivel solicitado no especificado",

            "tono":
                "warning",

            "nivel_candidato":
                nivel_candidato,

            "nivel_requerido":
                nivel_requerido,

            "nivel_explicito":
                False
        }


    if (
        nivel_requerido is not None
        and
        nivel_candidato >= nivel_requerido
    ):

        return {

            "codigo":
                "CUMPLE",

            "icono":
                "✓",

            "texto":
                "Cumple",

            "tono":
                "success",

            "nivel_candidato":
                nivel_candidato,

            "nivel_requerido":
                nivel_requerido,

            "nivel_explicito":
                True
        }

def preparar_conocimiento_v018842(
    match,
    requerida=None,
    candidato=None,
    oferta=None
):

    match = (
        match
        if isinstance(
            match,
            dict
        )
        else {}
    )


    requerida = str(
        requerida
        or
        match.get(
            "_requerida"
        )
        or
        match.get(
            "requerida"
        )
        or
        match.get(
            "nombre"
        )
        or
        ""
    ).strip()


    nombre_candidata = match.get(
        "candidato"
    )


    # ------------------------------------------------------------------
    # NIVEL REAL DEL CANDIDATO
    #
    # matching_habilidades()["nivel"] NO es necesariamente el nivel
    # 1-5 original. Es la salida de cumplimiento_nivel().
    #
    # Para mostrar "Tu nivel" debemos volver a candidato["habilidades"].
    # ------------------------------------------------------------------

    nivel_candidato = 0


    if (
        nombre_candidata
        and
        isinstance(
            candidato,
            dict
        )
    ):

        habilidades_candidato = candidato.get(
            "habilidades",
            {}
        )


        if isinstance(
            habilidades_candidato,
            dict
        ):

            valor = habilidades_candidato.get(
                nombre_candidata
            )


            if isinstance(
                valor,
                dict
            ):

                valor = (
                    valor.get(
                        "nivel"
                    )
                    or
                    valor.get(
                        "nivel_detectado"
                    )
                    or
                    0
                )


            try:

                nivel_candidato = int(
                    valor
                    or
                    0
                )


            except Exception:

                nivel_candidato = 0


    # ------------------------------------------------------------------
    # NIVEL REQUERIDO
    # ------------------------------------------------------------------

    try:

        nivel_requerido = int(
            match.get(
                "nivel_requerido",
                0
            )
            or
            0
        )


    except Exception:

        nivel_requerido = 0


    # ------------------------------------------------------------------
    # ¿FUE EXPLÍCITO?
    #
    # Se consulta directamente la estructura de la oferta porque
    # matching_habilidades() no transporta esa metadata.
    # ------------------------------------------------------------------

    nivel_explicito = False


    if (
        requerida
        and
        isinstance(
            oferta,
            dict
        )
    ):

        habilidades_oferta = oferta.get(
            "habilidades",
            {}
        )


        if isinstance(
            habilidades_oferta,
            dict
        ):

            datos_oferta = habilidades_oferta.get(
                requerida,
                {}
            )


            if isinstance(
                datos_oferta,
                dict
            ):

                nivel_explicito = bool(
                    datos_oferta.get(
                        "nivel_explicito",
                        False
                    )
                )


    # ------------------------------------------------------------------
    # ESTADO
    # ------------------------------------------------------------------

    if not nombre_candidata or nivel_candidato <= 0:

        estado = "Nivel no detectado"

        codigo_estado = "NO_DETECTADO"

        icono = "⚠"

        tono = "warning"


    elif not nivel_explicito:

        estado = "Nivel solicitado no especificado"

        codigo_estado = "NIVEL_NO_ESPECIFICADO"

        icono = "⚠"

        tono = "warning"


    elif nivel_candidato >= nivel_requerido:

        estado = "Cumple"

        codigo_estado = "CUMPLE"

        icono = "✓"

        tono = "success"


    else:

        estado = "Nivel inferior al solicitado"

        codigo_estado = "NIVEL_INFERIOR"

        icono = "⚠"

        tono = "warning"


    nivel_solicitado = (

        nombre_nivel_v018842(
            nivel_requerido,
            "No especificado"
        )

        if nivel_explicito

        else
        "No especificado"
    )


    return {

        "nombre":
            habilidad_nombre_amigable_v018842(
                requerida
            ),

        "habilidad_candidata":
            nombre_candidata,

        "nivel_solicitado":
            nivel_solicitado,

        "nivel_propio":
            nombre_nivel_v018842(
                nivel_candidato
            ),

        "estado":
            estado,

        "estado_codigo":
            codigo_estado,

        "estado_icono":
            icono,

        "tono":
            tono
    }


def render_conocimientos_compactos_v018841(
    evaluacion,
    candidato=None,
    oferta=None
):

    matches = (
        evaluacion.get(
            "matches",
            {}
        )
        if isinstance(
            evaluacion,
            dict
        )
        else {}
    )


    st.markdown(
        "### Conocimientos clave"
    )


    if not matches:

        st.caption(
            "No encontramos conocimientos específicos suficientes "
            "para comparar en esta oferta."
        )

        return


    # ------------------------------------------------------------------
    # ESTRUCTURA REAL:
    #
    # matches = {
    #     "Power BI": {...},
    #     "SQL": {...}
    # }
    #
    # La clave ES el conocimiento requerido.
    # ------------------------------------------------------------------

    if isinstance(
        matches,
        dict
    ):

        iterable = matches.items()


    elif isinstance(
        matches,
        list
    ):

        iterable = [

            (
                str(
                    indice + 1
                ),
                match
            )

            for indice, match
            in enumerate(
                matches
            )
        ]


    else:

        st.caption(
            "No fue posible interpretar los conocimientos de esta oferta."
        )

        return


    for requerida, match in iterable:

        conocimiento = preparar_conocimiento_v018842(
            match,
            requerida=requerida,
            candidato=candidato,
            oferta=oferta
        )


        nombre = conocimiento[
            "nombre"
        ]


        propio = conocimiento[
            "nivel_propio"
        ]


        solicitado = conocimiento[
            "nivel_solicitado"
        ]


        estado = conocimiento[
            "estado"
        ]


        icono = conocimiento[
            "estado_icono"
        ]


        codigo_estado = conocimiento[
            "estado_codigo"
        ]


        with st.container(
            border=True
        ):

            st.markdown(
                f"**{nombre}**"
            )


            col_a, col_b = st.columns(
                2
            )


            with col_a:

                st.caption(
                    "Nivel solicitado"
                )


                if solicitado == "No especificado":

                    st.markdown(
                        "⚠ **No especificado**"
                    )


                else:

                    st.markdown(
                        f"**{solicitado}**"
                    )


            with col_b:

                st.caption(
                    "Tu nivel"
                )

                st.markdown(
                    f"**{propio}**"
                )


            if codigo_estado == "CUMPLE":

                st.success(
                    f"{icono} {estado}"
                )


            elif codigo_estado == "NIVEL_NO_ESPECIFICADO":

                st.warning(
                    "⚠ Nivel solicitado no especificado"
                )


            else:

                st.warning(
                    f"{icono} {estado}"
                )

def fortalezas_tarjeta_v018903(
    evaluacion
):

    fortalezas = []


    # --------------------------------------------------------------
    # MATCHES DE CONOCIMIENTOS
    # --------------------------------------------------------------

    matches = evaluacion.get(
        "matches",
        {}
    )


    if isinstance(
        matches,
        dict
    ):

        for requerida, datos in matches.items():

            if not isinstance(
                datos,
                dict
            ):

                continue


            candidata = datos.get(
                "candidato"
            )


            try:

                similitud = float(
                    datos.get(
                        "similitud",
                        0
                    )
                    or
                    0
                )


            except Exception:

                similitud = 0


            if (
                candidata
                and
                similitud >= 70
            ):

                nombre = str(
                    requerida
                ).strip()


                if (
                    nombre
                    and
                    nombre
                    not in fortalezas
                ):

                    fortalezas.append(
                        nombre
                    )


    # --------------------------------------------------------------
    # EXPERIENCIA CONTEXTUAL
    # --------------------------------------------------------------

    detalle = evaluacion.get(
        "experiencia_contextual_detalle",
        {}
    )


    if isinstance(
        detalle,
        dict
    ):

        directa = float(
            detalle.get(
                "directa",
                0
            )
            or
            0
        )


        funcional = float(
            detalle.get(
                "funcional_relevante",
                0
            )
            or
            0
        )


        relacionada = float(
            detalle.get(
                "relacionada",
                0
            )
            or
            0
        )


        transferible = float(
            detalle.get(
                "transferible_alto_valor",
                0
            )
            or
            0
        )


        if directa > 0:

            fortalezas.append(
                "Experiencia directa"
            )


        elif funcional > 0:

            fortalezas.append(
                "Experiencia funcional relevante"
            )


        elif relacionada > 0:

            fortalezas.append(
                "Experiencia relacionada"
            )


        elif transferible > 0:

            fortalezas.append(
                "Experiencia transferible de alto valor"
            )


    # --------------------------------------------------------------
    # DEDUPLICAR
    # --------------------------------------------------------------

    salida = []


    for item in fortalezas:

        if item not in salida:

            salida.append(
                item
            )


    return salida[:5]

        st.subheader(
            "Perfil inteligente"
        )


        p1, p2, p3, p4 = st.columns(
            4
        )


        p1.metric(
            "Nombre",
            candidato[
                "nombre"
            ]
            or
            "No detectado"
        )


        p2.metric(
            "Profesión",
            candidato[
                "profesion"
            ]
            or
            "No detectada"
        )


        p3.metric(
            "Experiencia",
            (
                f"{candidato['experiencia']} años"

                if candidato[
                    "experiencia"
                ]
                is not None

                else
                "No detectada"
            )
        )


        p4.metric(
            "Ciudad",
            candidato[
                "ciudad"
            ]
            or
            "No detectada"
        )


        st.markdown(
            "### Habilidades"
        )


        for skill, nivel in candidato[
            "habilidades"
        ].items():

            st.write(
                f"**{skill}** — "
                f"{NOMBRE_NIVEL.get(nivel, nivel)}"
            )


    # --------------------------------------------------------------
    # ACTIVIDAD
    # --------------------------------------------------------------

    with tab_act:

        aprendizaje = perfil[
            "aprendizaje"
        ]


        a1, a2 = st.columns(
            2
        )


        a1.metric(
            "Interacciones",
            aprendizaje[
                "interacciones"
            ]
        )


        a2.metric(
            "Confianza aprendizaje",
            f"{aprendizaje['confianza']:.1f}%"
        )


        if aprendizaje[
            "habilitado"
        ]:

            st.success(
                "Aprendizaje conductual activo."
            )

        else:

            st.info(
                "Todavía se necesitan más interacciones."
            )


        st.markdown(
            "### Historial"
        )


        for evento in reversed(
            historial
        ):

            st.write(
                f"**#{evento['id']} "
                f"{evento['accion'].upper()}** · "
                f"{evento['oferta_nombre']} · "
                f"{evento['cargo']}"
            )


    # --------------------------------------------------------------
    # CUENTA
    # --------------------------------------------------------------

    with tab_cuenta:

        st.subheader(
            "Cuenta"
        )


        st.success(
            "Persistencia remota activa."
        )


        st.write(
            f"**Usuario:** @{usuario['username']}"
        )


        st.write(
            f"**ID:** {usuario['id']}"
        )


        st.write(
            f"**Creada:** {usuario['creado_en']}"
        )


        st.write(
            "**Fuente de verdad:** PostgreSQL / Supabase"
        )


        st.info(
            "Esta versión no importa automáticamente "
            "credenciales desde copias históricas."
        )


        st.divider()


        st.markdown(
            "### Cambiar contraseña"
        )


        with st.form(
            "cambiar_password"
        ):

            password_actual = st.text_input(
                "Contraseña actual",
                type="password"
            )


            password_nueva = st.text_input(
                "Nueva contraseña",
                type="password"
            )


            password_confirmar = st.text_input(
                "Confirmar nueva contraseña",
                type="password"
            )


            cambiar = st.form_submit_button(
                "Actualizar contraseña"
            )


        if cambiar:

            autenticado = autenticar_usuario(
                usuario[
                    "username"
                ],
                password_actual
            )


            if autenticado is None:

                st.error(
                    "La contraseña actual es incorrecta."
                )


            elif (
                password_nueva
                !=
                password_confirmar
            ):

                st.error(
                    "Las nuevas contraseñas no coinciden."
                )


            else:

                ok, mensaje = actualizar_password(
                    user_id,
                    password_nueva
                )


                if ok:

                    st.success(
                        "Contraseña actualizada correctamente."
                    )

                else:

                    st.error(
                        mensaje
                    )
